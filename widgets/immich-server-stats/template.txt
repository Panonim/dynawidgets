{{ $usageDisplayGiB := div (toFloat (toInt (add (mul $usageGiB 10.0) 0.5))) 10.0 }}
{{ $storage := .Subrequest "storage" }}
{{ $storageOK := and (ge $storage.Response.StatusCode 200) (le $storage.Response.StatusCode 299) }}
{{ $totalGiB := 0.0 }}
{{ $diskUsedGiB := 0.0 }}
{{ $diskUsedDisplayGiB := 0.0 }}
{{ $otherUsageGiB := 0.0 }}
{{ $percentage := 0.0 }}

  {{ $totalGiB = div (toFloat ($storage.JSON.Int "diskSizeRaw")) 1073741824.0 }}
  {{ $diskUsedGiB = div (toFloat ($storage.JSON.Int "diskUseRaw")) 1073741824.0 }}
  {{ $diskUsedDisplayGiB = div (toFloat (toInt (add (mul $diskUsedGiB 10.0) 0.5))) 10.0 }}
  {{ $otherUsageGiB = sub $diskUsedDisplayGiB $usageDisplayGiB }}
  {{ $percentage = $storage.JSON.Float "diskUsagePercentage" }}
{{ end }}

  {{ if $storageOK }}

      <div class="flex justify-between items-center" style="margin-bottom: 0.5rem;">

  {{ end }}
</div>
